#include <GL/glut.h>
#include <cmath>

float fillColor[3] = {0, 0, 1}, boundaryColor[3] = {1, 0, 0};
int method = 0; // 1 for flood fill, 2 for boundary fill

void setPixel(int x, int y) {
    glBegin(GL_POINTS);
    glVertex2i(x, y);
    glEnd();
    glFlush();
}

void getColor(int x, int y, float color[3]) {
    glReadPixels(x, y, 1, 1, GL_RGB, GL_FLOAT, color);
}

bool isSameColor(float a[3], float b[3]) {
    return fabs(a[0]-b[0]) < 0.01 && fabs(a[1]-b[1]) < 0.01 && fabs(a[2]-b[2]) < 0.01;
}

void floodFill(int x, int y, float oldColor[3]) {
    float color[3];
    getColor(x, y, color);
    if (isSameColor(color, oldColor)) {
        glColor3fv(fillColor); setPixel(x, y);
        floodFill(x+1, y, oldColor); floodFill(x-1, y, oldColor);
        floodFill(x, y+1, oldColor); floodFill(x, y-1, oldColor);
    }
}

void boundaryFill(int x, int y) {
    float color[3];
    getColor(x, y, color);
    if (!isSameColor(color, boundaryColor) && !isSameColor(color, fillColor)) {
        glColor3fv(fillColor); setPixel(x, y);
        boundaryFill(x+1, y); boundaryFill(x-1, y);
        boundaryFill(x, y+1); boundaryFill(x, y-1);
    }
}

void mouse(int btn, int state, int x, int y) {
    if (btn == GLUT_LEFT_BUTTON && state == GLUT_DOWN) {
        y = 480 - y;
        float current[3];
        getColor(x, y, current);
        if (method == 1) floodFill(x, y, current);
        else if (method == 2) boundaryFill(x, y);
    }
}

void keyboard(unsigned char key, int, int) {
    if (key == 'f') method = 1;
    else if (key == 'b') method = 2;
}

void menu(int option) { method = option; }

void polygon() {
    glClear(GL_COLOR_BUFFER_BIT);
    glLineWidth(3);
    glColor3fv(boundaryColor);
    glBegin(GL_LINE_LOOP);
    glVertex2i(150, 100); glVertex2i(300, 300);
    glVertex2i(450, 100);
    glEnd();
    glFlush();
}

void init() {
    glClearColor(1, 1, 1, 1);
    glColor3f(0, 0, 0);
    glPointSize(2);
    gluOrtho2D(0, 640, 0, 480);
}

int main(int argc, char** argv) {
    glutInit(&argc, argv);
    glutInitDisplayMode(GLUT_SINGLE | GLUT_RGB);
    glutInitWindowSize(640, 480);
    glutInitWindowPosition(100, 100);
    glutCreateWindow("Flood & Boundary Fill");
    init();
    glutDisplayFunc(polygon);
    glutMouseFunc(mouse);
    glutKeyboardFunc(keyboard);
    glutCreateMenu(menu);
    glutAddMenuEntry("Flood Fill", 1);
    glutAddMenuEntry("Boundary Fill", 2);
    glutAttachMenu(GLUT_RIGHT_BUTTON);
    glutMainLoop();
    return 0;
}
